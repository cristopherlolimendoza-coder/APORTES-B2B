  function rebuildKamFilter() {
    const select = $('fExec');
    if (!select) return;
    const previous = canonicalExecutive(select.value);
    const names = [...new Set(rows()
      .map(row => canonicalExecutive(row.ejecutivo, row.ruc))
      .filter(Boolean))]
      .sort((a, b) => a.localeCompare(b, 'es'));
    select.innerHTML = '<option value="">Todos</option>' + names
      .map(name => `<option value="${esc(name)}">${esc(name)}</option>`)
      .join('');
    select.value = names.includes(previous) ? previous : '';
  }

  }
