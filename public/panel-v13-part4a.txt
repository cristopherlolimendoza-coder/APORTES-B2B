13-view="accounts">Cuentas</button>`;

  }

  function setView(view) {
    document.querySelectorAll('.view').forEach(section => section.classList.remove('on'));
    document.querySelectorAll('[data-v13-view]').forEach(button => {
      button.classList.toggle('on', button.dataset.v13View === view);
    });
    $(view)?.classList.add('on');
    scheduleRender();
  }
