nction aggregate(list) {
    const output = {
      clients: new Set(),
      ops: list.length,
      won: 0,
      lost: 0,
      open: 0,
      refUnits: 0,
      wonUnits: 0,
      approved: 0,
      cost: 0,
      pending: 0,
      released: 0
    };
    list.forEach(row => {
      const info = metrics(row);
      output.clients.add(row.ruc || row.cliente);
      output.refUnits += info.refUnits;
      output.wonUnits += info.wonUnits;
      output.approved += info.approved;
      output.cost += info.cost;
      output.pending += info.pending;
      output.released += info.released;
      if (info.state === 'Ganada') output.won += 1;
      else if (info.state === 'Perdida') output.lost += 1;
      else output.open += 1;
    });
    output.closed = output.won + output.lost;
    output.winRate = output.closed ? (output.won / output.closed) * 100 : 0;
    return output;
  }

  function monthData(list) {
    return MONTHS
      .filter(month => list.some(row => row.mesOrigen === month))
      .map(month => ({
        month,
        label: MONTH_NAME[month] || month,
        rows: list.filter(row => row.mesOrigen === month),
        data: aggregate(list.filter(row => row.mesOrigen === month))
      }));
  }

  function kamData(list) {
    const map = new Map();
    list.forEach(row => {
      const name = canonicalExecutive(row.ejecutivo, row.ruc) || 'Sin KAM';
      const info = metrics(row);
      const current = map.get(name) || {
        name,
        ops: 0,
        won: 0,
        lost: 0,
        open: 0,
        refUnits: 0,
        wonUnits: 0,
        approved: 0,
        cost: 0,
        pending: 0
      };
      current.ops += 1;
      current.refUnits += info.refUnits;
      current.wonUnits += info.wonUnits;
      current.approved += info.approved;
      current.cost += info.cost;
      current.pending += info.pending;
      if (info.state === 'Ganada') current.won += 1;
      else if (info.state === 'Perdida') current.lost += 1;
      else current.open += 1;
      map.set(name, current);
    });
    return [...map.values()];
  }

  function modelData(list) {
    const map = new Map();
    list.forEach(row => {
      parseModels(row).forEach(item => {
        const name = item.modelo || 'Modelo sin definir';
        const current = map.get(name) || {
          name,
          accounts: new Set(),
          refUnits: 0,
          wonUnits: 0,
          approved: 0,
          cost: 0,
          pending: 0
        };
        const potential = item.unidades * item.aporteUnitario;
        const confirmed = item.estado === 'Ganada' ? item.unidadesGanadas * item.aporteUnitario : 0;
        const pending = (item.estado === 'En curso' || item.estado === 'Sin definir') ? potential : 0;
        current.accounts.add(String(row.id));
        current.refUnits += item.unidades;
        current.wonUnits += item.estado === 'Ganada' ? item.unidadesGanadas : 0;
        current.approved += potential;
        current.cost += confirmed;
        current.pending += pending;
        map.set(name, current);
      });
    });
    return [...map.values()];
  }

  function kpi(label, value, detail = '', tone = '') {
    return `<div class="v13-kpi ${tone}"><span>${esc(label)}</span><b>${value}</b><small>${esc(detail)}</small></div>`;
  }

  function badge(state) {
    const tone = state === 'Ganada' ? 'good' : state === 'Perdida' ? 'bad' : 'open';
    return `<span class="v13-badge ${tone}">${esc(state)}</span>`;
  }

  function empty(message = 'Sin datos para el filtro seleccionado.') {
    return `<div class="v13-empty">${esc(message)}</div>`;
  }
